class Reactivo:
    def __init__(self, codigo, nombre, cantidad, unidad):
        self.codigo = codigo
        self.nombre = nombre
        self.cantidad = cantidad
        self.unidad = unidad

    def ingresar(self, cantidad):
        if cantidad > 0:
            self.cantidad += cantidad
            print(f"-> Ingresados {cantidad} {self.unidad} de {self.nombre}.")
        else:
            print("-> Error: La cantidad a ingresar debe ser estrictamente positiva.")

    def consumir(self, cantidad):
        if cantidad <= 0:
            print("-> Error: La cantidad a consumir debe ser estrictamente positiva.")
        elif cantidad > self.cantidad:
            print(f"-> Error: Stock insuficiente de {self.nombre}. Stock actual: {self.cantidad} {self.unidad}.")
        else:
            self.cantidad -= cantidad
            print(f"-> Consumidos {cantidad} {self.unidad} de {self.nombre}.")

    def consultar_stock(self):
        return f"[{self.codigo}] {self.nombre}: {self.cantidad} {self.unidad}"

    r1 = Reactivo("R001", "Ácido Sulfúrico", 500, "ml")
    r2 = Reactivo("R002", "Hidróxido de Sodio", 1000, "g")
    
    print(r1.consultar_stock())
    r1.ingresar(200)
    r1.consumir(800)  # Intento fallido por falta de stock
    r1.consumir(300)  # Consumo exitoso
    r1.consumir(-50)  # Intento fallido por cantidad inválida
    
    print("Stock Final:")
    print(r1.consultar_stock())
    print(r2.consultar_stock())

class CuentaViaje:
    def __init__(self, titular, saldo_inicial, tarifa):
        self.titular = titular
        self.saldo = max(0.0, saldo_inicial)
        self.tarifa = tarifa

    def recargar(self, monto):
        if monto > 0:
            self.saldo += monto
            print(f"[{self.titular}] Recarga exitosa de S/{monto:.2f}.")
        else:
            print(f"[{self.titular}] Error: El monto de recarga debe ser positivo.")

    def pagar_viaje(self):
        if self.saldo >= self.tarifa:
            self.saldo -= self.tarifa
            print(f"[{self.titular}] Viaje pagado. Tarifa: S/{self.tarifa:.2f}.")
            return True
        else:
            print(f"[{self.titular}] Saldo insuficiente. Requiere S/{self.tarifa:.2f}, tiene S/{self.saldo:.2f}.")
            return False

    def consultar_saldo(self):
        return f"Cuenta {self.titular} - Saldo actual: S/{self.saldo:.2f}"

class Paciente:
    def __init__(self, identificacion, nombre):
        self.identificacion = identificacion
        self.nombre = nombre

    def __str__(self):
        return f"{self.nombre} (DNI: {self.identificacion})"

class Turno:
    def __init__(self, paciente, fecha_hora):
        self.paciente = paciente
        self.fecha_hora = fecha_hora

    def __str__(self):
        return f"[{self.fecha_hora}] - Paciente: {self.paciente}"

class Agenda:
    def __init__(self):
        self.turnos = []

    def registrar_turno(self, paciente, fecha_hora):
        for turno in self.turnos:
            if turno.fecha_hora == fecha_hora:
                print(f"Error: El horario {fecha_hora} ya se encuentra ocupado.")
                return False
        
        nuevo_turno = Turno(paciente, fecha_hora)
        self.turnos.append(nuevo_turno)
        print(f"Éxito: Turno registrado para {paciente.nombre} a las {fecha_hora}.")
        return True

    def listar_turnos_paciente(self, identificacion):
        print(f"\n--- Turnos del paciente DNI: {identificacion} ---")
        encontrados = [t for t in self.turnos if t.paciente.identificacion == identificacion]
        
        if not encontrados:
            print("No hay turnos registrados.")
        else:
            for t in encontrados:
                print(t)

class Sensor:
    def __init__(self, identificador, tipo):
        self.identificador = identificador
        self.tipo = tipo
        self.lecturas = []

    def registrar_lectura(self, valor):
        self.lecturas.append(valor)

    def calcular_promedio(self):
        if not self.lecturas:
            return 0.0
        return sum(self.lecturas) / len(self.lecturas)

    def estado_alerta(self, umbral):
        return self.calcular_promedio() > umbral

    def __str__(self):
        return f"Sensor {self.identificador} ({self.tipo}) - Promedio de lecturas: {self.calcular_promedio():.2f}"
